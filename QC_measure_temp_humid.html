📋 Supabase SQL (테이블 생성) 
CREATE TABLE IF NOT EXISTS qc_measure_log (
  id BIGSERIAL PRIMARY KEY,
  check_date DATE NOT NULL,
  check_time TIME NOT NULL,
  inspector TEXT NOT NULL,
  equipment TEXT NOT NULL,
  items_json JSONB,
  ng_count INT DEFAULT 0,
  result TEXT DEFAULT 'OK',
  memo TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_qc_measure_log_eq_date
  ON qc_measure_log(equipment, check_date DESC);
ALTER TABLE qc_measure_log DISABLE ROW LEVEL SECURITY;
GRANT ALL ON qc_measure_log TO anon, authenticated;
GRANT USAGE, SELECT ON SEQUENCE qc_measure_log_id_seq TO anon, authenticated;
